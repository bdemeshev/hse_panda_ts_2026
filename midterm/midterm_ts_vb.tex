  \[
  \begin{cases}

  \ell_t = \ell_{t-1} + 0.4 b_{t-1} + 0.1 u_t \\
  y_t = \ell_{t-1} + 0.4 b_{t-1} + u_t \\
  \end{cases}
  \]
  with $\ell_{100} = 30$ and $b_{100} = 2$.
\begin{enumerate}
  \item {[5]} Find the 95\% predictive interval for $y_{102}$.
  \item {[5]} Find conditional probability $\P(y_{102} > 40 \mid \ell_{100}, b_{100})$ in terms of the standard normal distribution function.
\end{enumerate}

\item   The semi-annual $y_t$ is modelled by $ETS(ANA)$ process:    
    \[

        \ell_t = \ell_{t-1}  + 0.5 u_t \\
        y_t = \ell_{t-1} + s_{t-2} + u_t \\
    \end{cases}    
    \]
with $\ell_0 = 20$, $s_0 = 5$ and $s_{-1} = -5$.

\begin{enumerate}
    \item {[4]} Find best point forecasts for $y_{100}$ and $y_{101}$.

\end{enumerate}
