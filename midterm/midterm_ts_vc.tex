\begin{enumerate}
\item   The $ETS(AAdN)$ model is given by the system
  \[
  \begin{cases}
  u_t  \sim \cN(0; 16) \\
  b_t = 0.5 b_{t-1} + 0.8 u_t \\
  \ell_t = \ell_{t-1} + 0.5 b_{t-1} + 0.8 u_t \\
  y_t = \ell_{t-1} + 0.5 b_{t-1} + u_t \\
  \end{cases}
  \]
  with $\ell_{100} = 30$ and $b_{100} = 2$.
\begin{enumerate}
  \item {[5]} Find the 95\% predictive interval for $y_{102}$.
  \item {[5]} Find conditional probability $\P(y_{102} < 40  \mid \ell_{100}, b_{100})$ in terms of the standard normal distribution function.
\end{enumerate}
\item   The semi-annual $y_t$ is modelled by $ETS(AAA)$ process:    
    \[
    \begin{cases}
        u_t \sim \cN(0; 16) \\
        b_t = b_{t-1} + 0.8 u_t \\
        s_t = s_{t-2} + 0.1 u_t \\
        \ell_t = \ell_{t-1} + b_{t-1} + 0.5 u_t \\
        y_t = \ell_{t-1} + b_{t-1} + s_{t-2} + u_t \\
    \end{cases}    
    \]
with $\ell_0 = 20$, $b_0 = 1$, $s_0 = 5$ and $s_{-1} = -5$.

Decompose first two observations $y_1 = 25$ and $y_2 = 20$ into trend, seasonal and error components.

\item Consider $MA(1)$ process $y_t = 3 + u_t + 4 u_{t-1}$, where $u_t$ is a white noise process with variance $10$.
\begin{enumerate}
    \item {[5]} Find $\E(y_t)$, $\Var(y_t)$, $\Cov(y_t, y_s)$.
    \item {[5]} Consider and independent copy $(y'_t)$ of $(y_t)$ (all expected value, variance and autocovariances are the same).
    If possible classify $s_t = y'_t + y_{t-1}$ as $MA(q)$ process. 
\end{enumerate}

\item Consider the equation
\(
y_t = 0.5 y_{t-1} + u_t + 0.4 u_{t-1},
\)
where \( u_t \) is a white noise process with variance $\sigma^2$.
\begin{enumerate}
    \item {[1]} How many non-stationary solutions does this equation have?
    \item {[3]} How many stationary solutions of $MA(\infty)$ form with respect to $(u_t)$ does this equation have? Find all solutions, if possible.
    \item {[2]} Rewrite all stationary solutions in $AR(\infty)$ form with respect to $(u_t)$, if possible.
    \item {[4]} Find $\Var(y_t)$ for all stationary solutions.
\end{enumerate}

\item Consider the stationary solution of the equation
\[
y_t = 0.8 y_{t-1} - 0.5 y_{t-2} + u_t, 
\]
where $(u_t)$ is a white noise.
\begin{enumerate}
    \item {[3]} Check the existence of stationary solution.
    \item {[4]} Find the values of ACF \( \rho_1, \rho_2, \rho_3 \).
    \item {[3]} Find the values of PACF \( \phi_{11}, \phi_{22}, \phi_{33}\).
\end{enumerate}

\item You know that $y_t$ is a stationary $ARMA(p, q)$ process. 
\begin{enumerate}
    \item Is it possible that autocorrelation $\rho_{10} = 0.5$ but all other autocorrelations ($\rho_k$ for $k \geq 1$ and $k\neq 10$) are zero?
    If possible provide an equation of such a process. 

    \item Is it possible that partial autocorrelation $\phi_{10, 10} = 0.5$ but all other partial autocorrelations ($\phi_{k,k}$ for $k \geq 1$ and $k\neq 10$) are zero?
    If possible provide an equation of such a process. 
\end{enumerate}

\end{enumerate}
